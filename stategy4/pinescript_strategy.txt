//@version=6
strategy("Adaptive SuperSmoother Strategy", overlay=true,
     initial_capital=10000, currency=currency.USD,
     default_qty_type=strategy.fixed, default_qty_value=10000,
     commission_type=strategy.commission.percent, commission_value=0.002,
     slippage=5, margin_long=3.33, margin_short=3.33,
     pyramiding=0, calc_on_every_tick=false, process_orders_on_close=false)

// ============================ INPUTS ============================
// --- Indicator settings (same defaults as the original) ---
float src          = input.source(close, "Source", group="Adaptive MA Settings")
int   maMin        = input.int(14, "Minimum MA Period", minval=1, group="Adaptive MA Settings", inline="A1")
int   maMax        = input.int(50, "Maximum MA Period", minval=2, group="Adaptive MA Settings", inline="A1")
int   volLen       = input.int(9, "Volatility Period", minval=2, group="Adaptive MA Settings")
int   smoothLength = input.int(3, "SuperSmoother Length", minval=3, group="Filter Settings")
int   erLength     = input.int(10, "Efficiency Length", minval=2, group="Efficiency Settings")
float fastBias     = input.float(0.30, "Minimum Fast Weight", minval=0.0, maxval=1.0, step=0.05, group="Efficiency Settings")
int   noiseLength  = input.int(20, "Noise Length", minval=2, group="Signal Settings", inline="S1")
float noiseMult    = input.float(0.60, "Noise Multiplier", minval=0.0, step=0.05, group="Signal Settings", inline="S1")

// --- Strategy settings ---
string tradeDir    = input.string("Long & Short", "Trade Direction", options=["Long & Short", "Long Only", "Short Only"], group="Strategy")
bool   useSL       = input.bool(true, "Use ATR Stop Loss", group="Risk")
bool   useTP       = input.bool(false, "Use ATR Take Profit", group="Risk")
int    atrLen      = input.int(14, "ATR Length", minval=1, group="Risk")
float  atrSL       = input.float(2.0, "Stop Loss (x ATR)", minval=0.1, step=0.1, group="Risk")
float  atrTP       = input.float(3.0, "Take Profit (x ATR)", minval=0.1, step=0.1, group="Risk")
string sizeMode    = input.string("Fixed Units", "Position Sizing", options=["Fixed Units", "Risk % of Equity"], group="Risk")
float  fixedQty    = input.float(10000, "Fixed Units (10000 = 0.1 lot)", minval=1, group="Risk")
float  riskPct     = input.float(1.0, "Risk % per Trade", minval=0.1, maxval=10, step=0.1, group="Risk")

// --- Date range ---
bool  useDate      = input.bool(false, "Limit Backtest Date Range", group="Date Range")
int   startDate    = input.time(timestamp("01 Jan 2023 00:00 +0000"), "Start", group="Date Range")
int   endDate      = input.time(timestamp("31 Dec 2030 23:59 +0000"), "End", group="Date Range")

// --- Visuals ---
bool  showSignals  = input.bool(true, "Show Signals", group="Visualization")
bool  colorBars    = input.bool(true, "Color Bars", group="Visualization")
color longColor    = input.color(color.rgb(57, 255, 20), "Bullish Color", group="Color Settings", inline="COL1")
color shortColor   = input.color(color.rgb(138, 43, 226), "Bearish Color", group="Color Settings", inline="COL1")

// ========================== CALCULATIONS ==========================
float deltaP = src - src[1]
float s      = ta.stdev(deltaP, volLen)
float sMean  = ta.sma(s, volLen)

float sMin = sMean * 0.25
float sMax = sMean * 1.75

float volatilityRange    = sMax - sMin
float volatilityPosition = volatilityRange > 0.0 ? math.max(0.0, math.min(1.0, (s - sMin) / volatilityRange)) : 0.0

float adaptiveLength = s <= sMin ? maMax : s >= sMax ? maMin : maMax - (maMax - maMin) * volatilityPosition
float alpha          = 2.0 / (adaptiveLength + 1.0)

var float adaptiveMA = na
adaptiveMA := na(adaptiveMA[1]) ? src : adaptiveMA[1] + alpha * (src - adaptiveMA[1])

float a1 = math.exp(-1.414 * math.pi / smoothLength)
float b1 = 2.0 * a1 * math.cos(1.414 * math.pi / smoothLength)
float c2 = b1
float c3 = -a1 * a1
float c1 = 1.0 - c2 - c3

var float smoothMA = na
smoothMA := na(smoothMA[1]) or na(smoothMA[2]) ? adaptiveMA : c1 * (adaptiveMA + nz(adaptiveMA[1], adaptiveMA)) * 0.5 + c2 * smoothMA[1] + c3 * smoothMA[2]

float direction = math.abs(src - nz(src[erLength], src))
float noise     = 0.0

for i = 0 to erLength - 1
    noise += math.abs(nz(src[i], src) - nz(src[i + 1], src))

float efficiency = noise > 0.0 ? direction / noise : 0.0
efficiency := math.max(0.0, math.min(1.0, efficiency))

float fastWeight = fastBias + (1.0 - fastBias) * efficiency
float slowWeight = 1.0 - fastWeight
float trend      = adaptiveMA * fastWeight + smoothMA * slowWeight

float slope      = trend - trend[1]
float slopeNoise = ta.rma(math.abs(slope), noiseLength)
float threshold  = slopeNoise * noiseMult

// ============================ SIGNALS ============================
bool isLong  = slope > threshold
bool isShort = slope < -threshold

var int SQ = 0

if isLong
    SQ := 1
else if isShort
    SQ := -1

bool longSignal  = SQ == 1 and SQ[1] != 1
bool shortSignal = SQ == -1 and SQ[1] != -1

// ============================ STRATEGY ============================
bool inRange   = not useDate or (time >= startDate and time <= endDate)
bool allowLong  = tradeDir != "Short Only"
bool allowShort = tradeDir != "Long Only"

float atrVal = ta.atr(atrLen)
var float entryATR = na

if longSignal or shortSignal
    entryATR := atrVal

// Position size
float stopDist = atrSL * atrVal
float qtyRisk  = stopDist > 0 ? math.max(1.0, math.floor(strategy.equity * riskPct / 100.0 / stopDist)) : fixedQty
float qty      = sizeMode == "Fixed Units" ? fixedQty : qtyRisk

// Entries (signal is confirmed on bar close, order fills on next bar open)
if longSignal and inRange
    if allowLong
        strategy.entry("Long", strategy.long, qty=qty)
    else
        strategy.close_all(comment="Trend flip")

if shortSignal and inRange
    if allowShort
        strategy.entry("Short", strategy.short, qty=qty)
    else
        strategy.close_all(comment="Trend flip")

// Optional ATR stop / take profit (the opposite signal still reverses the position)
if strategy.position_size > 0 and not na(entryATR)
    float slL = useSL ? strategy.position_avg_price - atrSL * entryATR : na
    float tpL = useTP ? strategy.position_avg_price + atrTP * entryATR : na
    strategy.exit("Long Exit", "Long", stop=slL, limit=tpL)

if strategy.position_size < 0 and not na(entryATR)
    float slS = useSL ? strategy.position_avg_price + atrSL * entryATR : na
    float tpS = useTP ? strategy.position_avg_price - atrTP * entryATR : na
    strategy.exit("Short Exit", "Short", stop=slS, limit=tpS)

// Close everything at the end of the chosen date range
if useDate and time > endDate
    strategy.close_all(comment="End of range")

// ============================= VISUALS =============================
color col    = SQ == 1 ? longColor : shortColor
color colors = colorBars ? col : na

plot(trend, title="Adaptive SuperSmoother Trend", color=col, linewidth=2)

plotshape(showSignals and longSignal, title="Long Signal", style=shape.triangleup, location=location.belowbar, size=size.small, color=longColor, text="LONG", textcolor=color.white)
plotshape(showSignals and shortSignal, title="Short Signal", style=shape.triangledown, location=location.abovebar, size=size.small, color=shortColor, text="SHORT", textcolor=color.white)

barcolor(colors, title="Bar Color")

// ============================= ALERTS =============================
alertcondition(longSignal, title="Long Signal", message="Adaptive SuperSmoother Trend Long Signal on {{ticker}}")
alertcondition(shortSignal, title="Short Signal", message="Adaptive SuperSmoother Trend Short Signal on {{ticker}}")
