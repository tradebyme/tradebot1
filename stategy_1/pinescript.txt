//@version=5
strategy("London/NY Momentum Pullback [Strategy 1]", 
         shorttitle="Strat1 Momentum Pullback", 
         overlay=true, 
         initial_capital=2000, 
         default_qty_type=strategy.cash, 
         default_qty_value=2000, 
         currency=currency.USD, 
         commission_type=strategy.commission.percent, 
         commission_value=0.002, 
         slippage=1,
         process_orders_on_close=false)

// =============================================================================
// 1. USER INPUTS & STRATEGY SETTINGS
// =============================================================================

// --- Moving Averages ---
group_ma = "Moving Average Trend Filters"
int ema20Len  = input.int(20,  "Fast EMA (Pullback Base)", group=group_ma)
int ema50Len  = input.int(50,  "Medium EMA (Trend Filter)", group=group_ma)
int ema100Len = input.int(100, "Macro EMA (Baseline Filter)", group=group_ma)

// --- ATR & Volatility Stops ---
group_atr = "ATR & Dynamic Volatility Stops"
int atrLen            = input.int(14, "ATR Period", group=group_atr)
string atrMethod      = input.string("SMA (Exact Python Bot)", "ATR Method", options=["SMA (Exact Python Bot)", "RMA (Wilder's Standard)"], group=group_atr, tooltip="Python backtest used 14-period SMA of True Range. RMA is standard TradingView ATR.")
float slAtrMult       = input.float(1.5, "Stop Loss Multiplier (x ATR)", step=0.1, group=group_atr)
float tpAtrMult       = input.float(3.0, "Take Profit Multiplier (x ATR)", step=0.1, group=group_atr, tooltip="Default 3.0 gives an institutional 1:2 Risk-Reward ratio with 1.5x SL.")
float pullbackTol     = input.float(0.75, "Pullback Zone Distance (x ATR)", step=0.05, group=group_atr, tooltip="Price low/high must be within EMA20 +/- (pullbackTol * ATR).")
int pullbackLookback  = input.int(3, "Pullback Lookback Candles", minval=1, maxval=10, group=group_atr)

// --- RSI Settings ---
group_rsi = "RSI Momentum Filter"
int rsiLen          = input.int(14, "RSI Length", group=group_rsi)
float rsiBullMin    = input.float(40.0, "Bullish RSI Lower Band", group=group_rsi)
float rsiBullMax    = input.float(62.0, "Bullish RSI Upper Band", group=group_rsi)
float rsiBearMin    = input.float(38.0, "Bearish RSI Lower Band", group=group_rsi)
float rsiBearMax    = input.float(60.0, "Bearish RSI Upper Band", group=group_rsi)

// --- Session & Time Filters ---
group_session = "Session & Weekend Filter (UTC)"
bool useSessionFilter = input.bool(true, "Enable Session Filter", group=group_session)
int sessionStartHour  = input.int(7, "Session Start Hour (UTC)", minval=0, maxval=23, group=group_session, tooltip="London open starts 07:00 UTC")
int sessionEndHour    = input.int(17, "Session End Hour (UTC)", minval=0, maxval=23, group=group_session, tooltip="NY overlap ends 17:00 UTC")
bool useWeekendFilter = input.bool(true, "Filter Friday Post-Close & Weekends", group=group_session)

// --- Prop Firm Risk & Capital Protection ---
group_risk = "Prop Firm Challenge & Circuit Breakers"
bool enablePropLimits  = input.bool(true, "Enforce Prop Firm Rules", group=group_risk)
float propCapital      = input.float(2000.0, "Challenge Capital ($)", group=group_risk)
string sizingMode      = input.string("Dynamic Dollar Risk ($25/trade)", "Sizing Mode", options=["Dynamic Dollar Risk ($25/trade)", "Risk % of Equity", "Fixed Units"], group=group_risk)
float riskPerTradeUSD  = input.float(25.0, "Risk Per Trade ($)", group=group_risk, tooltip="1.25% on $2,000 = $25.00 fixed risk per trade.")
float riskPerTradePct  = input.float(1.25, "Risk Per Trade (%)", group=group_risk)
float fixedUnits       = input.float(10000.0, "Fixed Position Units", group=group_risk)
int maxDailyLosses     = input.int(2, "Max Daily Losses (Circuit Breaker)", group=group_risk, tooltip="Halts trading if 2 losses occur in 1 day (0% drawdown failure in 2-yr test).")
float circuitBreakerUSD= input.float(70.0, "Daily Loss Circuit Breaker ($)", group=group_risk, tooltip="Stops taking trades if daily loss hits -$70 (-3.5%).")
int maxDailyTrades     = input.int(3, "Max Daily Trades", group=group_risk)
bool lockOnTarget      = input.bool(true, "Halt Trading When Target Hit (+8%)", group=group_risk)
float targetProfitUSD  = input.float(160.0, "Target Profit ($)", group=group_risk, tooltip="+8.0% = +$160.00 on $2,000.")
bool lockOnMaxDD       = input.bool(true, "Halt Trading When Max Loss Hit (-8%)", group=group_risk)
float maxDrawdownUSD   = input.float(160.0, "Max Total Drawdown ($)", group=group_risk, tooltip="-8.0% = -$160.00 on $2,000.")

// --- Display & Dashboard ---
group_ui = "Visuals & Chart Dashboard"
bool showEmas         = input.bool(true, "Show EMAs on Chart", group=group_ui)
bool showSessionBg    = input.bool(true, "Highlight London/NY Session", group=group_ui)
bool showSLTP         = input.bool(true, "Plot Active SL / TP Levels", group=group_ui)
bool showDashboard    = input.bool(true, "Display Prop Dashboard Table", group=group_ui)
string dashPosition   = input.string(position.top_right, "Dashboard Position", options=[position.top_right, position.bottom_right, position.top_left, position.bottom_left], group=group_ui)

// =============================================================================
// 2. INDICATOR CALCULATIONS
// =============================================================================

// Exponential Moving Averages
float ema20  = ta.ema(close, ema20Len)
float ema50  = ta.ema(close, ema50Len)
float ema100 = ta.ema(close, ema100Len)

// Average True Range (ATR)
float trVal  = ta.tr(true)
float atrVal = (atrMethod == "SMA (Exact Python Bot)") ? ta.sma(trVal, atrLen) : ta.atr(atrLen)

// Relative Strength Index (RSI)
float rsiVal = ta.rsi(close, rsiLen)

// Pullback Range
float minRecentLow  = ta.lowest(low, pullbackLookback)
float maxRecentHigh = ta.highest(high, pullbackLookback)

// =============================================================================
// 3. SESSION & TIME FILTERS
// =============================================================================

int utcHour = hour(time, "UTC")
int utcDay  = dayofweek(time, "UTC")

bool isSessionHours = (utcHour >= sessionStartHour and utcHour < sessionEndHour)
// Friday after 21:00 UTC, Saturday all day, Sunday before 21:00 UTC
bool isWeekend = (utcDay == dayofweek.saturday) or (utcDay == dayofweek.sunday and utcHour < 21) or (utcDay == dayofweek.friday and utcHour >= 21)
bool inSession = useSessionFilter ? (isSessionHours and not (useWeekendFilter and isWeekend)) : true

// =============================================================================
// 4. PROP FIRM CHALLENGE & CIRCUIT BREAKER ENGINE
// =============================================================================

// Detect UTC Midnight Rollover
bool isNewDay = dayofmonth(time, "UTC") != dayofmonth(time[1], "UTC")

var int   dailyTradesCount = 0
var int   dailyLossesCount = 0
var float dailyPnL         = 0.0

if isNewDay
    dailyTradesCount := 0
    dailyLossesCount := 0
    dailyPnL         := 0.0

// Increment opened trades count
var int prevOpenTrades = 0
if strategy.opentrades > prevOpenTrades
    dailyTradesCount += (strategy.opentrades - prevOpenTrades)
prevOpenTrades := strategy.opentrades

// Track closed trades and daily realized PnL / Losses
var int prevClosedTrades = 0
if strategy.closedtrades > prevClosedTrades
    int numClosed = strategy.closedtrades - prevClosedTrades
    for i = 0 to numClosed - 1
        int tradeIdx = strategy.closedtrades - 1 - i
        float profit = strategy.closedtrades.profit(tradeIdx)
        dailyPnL += profit
        if profit < 0
            dailyLossesCount += 1
    prevClosedTrades := strategy.closedtrades

// Circuit Breakers Evaluation
bool dailyLossBreaker  = enablePropLimits and (dailyLossesCount >= maxDailyLosses or dailyPnL <= -circuitBreakerUSD)
bool dailyMaxTradesHit = enablePropLimits and (dailyTradesCount >= maxDailyTrades)
bool targetPassed      = enablePropLimits and lockOnTarget and (strategy.netprofit >= targetProfitUSD)
bool maxDrawdownHit    = enablePropLimits and lockOnMaxDD and (strategy.netprofit <= -maxDrawdownUSD)

bool canTradeToday     = (not enablePropLimits) or (not dailyLossBreaker and not dailyMaxTradesHit and not targetPassed and not maxDrawdownHit)

// =============================================================================
// 5. STRATEGY ENTRY CONDITIONS
// =============================================================================

// Valid ATR check
bool atrOk = not na(atrVal) and atrVal > 0.00005

// Bullish (BUY) Setup:
// 1. Trend: EMA 20 > EMA 50 and Close > EMA 100
// 2. Pullback: Low in last 3 candles touched within 0.75 ATR of EMA 20
// 3. RSI: Healthy momentum zone (40 to 62)
// 4. Trigger: Bullish green candle (Close > Open) and Close > EMA 20
bool is_bull_trend        = (ema20 > ema50) and (close > ema100)
bool pulled_back_support  = minRecentLow <= (ema20 + pullbackTol * atrVal)
bool rsi_bull             = (rsiVal >= rsiBullMin) and (rsiVal <= rsiBullMax)
bool trigger_bull         = (close > open) and (close > ema20)

bool buySignal = is_bull_trend and pulled_back_support and rsi_bull and trigger_bull and inSession and atrOk

// Bearish (SELL) Setup:
// 1. Trend: EMA 20 < EMA 50 and Close < EMA 100
// 2. Pullback: High in last 3 candles touched within 0.75 ATR of EMA 20
// 3. RSI: Healthy momentum zone (38 to 60)
// 4. Trigger: Bearish red candle (Close < Open) and Close < EMA 20
bool is_bear_trend        = (ema20 < ema50) and (close < ema100)
bool pulled_back_resist   = maxRecentHigh >= (ema20 - pullbackTol * atrVal)
bool rsi_bear             = (rsiVal >= rsiBearMin) and (rsiVal <= rsiBearMax)
bool trigger_bear         = (close < open) and (close < ema20)

bool sellSignal = is_bear_trend and pulled_back_resist and rsi_bear and trigger_bear and inSession and atrOk

// =============================================================================
// 6. POSITION SIZING & EXECUTION
// =============================================================================

// Dynamic Position Sizing (Ensures exact $25 loss if SL hits)
float riskAmount = fixedUnits
if sizingMode == "Dynamic Dollar Risk ($25/trade)"
    riskAmount := riskPerTradeUSD
else if sizingMode == "Risk % of Equity"
    riskAmount := strategy.equity * (riskPerTradePct / 100.0)

float stopDistance = slAtrMult * atrVal
float calculatedUnits = (stopDistance > 0 and sizingMode != "Fixed Units") ? math.round(riskAmount / stopDistance) : fixedUnits
calculatedUnits := math.max(1.0, calculatedUnits)

var float tradeSL = na
var float tradeTP = na

// Long Entry
if buySignal and canTradeToday and strategy.position_size == 0
    float slTicks = math.round((slAtrMult * atrVal) / syminfo.mintick)
    float tpTicks = math.round((tpAtrMult * atrVal) / syminfo.mintick)
    strategy.entry("Long", strategy.long, qty=calculatedUnits, comment="BUY [Strat 1]")
    strategy.exit("Exit Long", "Long", loss=slTicks, profit=tpTicks, comment_loss="SL Hit", comment_profit="TP Hit")
    tradeSL := close - (slAtrMult * atrVal)
    tradeTP := close + (tpAtrMult * atrVal)
    alert("Strategy 1 BUY Signal triggered on " + syminfo.ticker, alert.freq_once_per_bar_close)

// Short Entry
if sellSignal and canTradeToday and strategy.position_size == 0
    float slTicks = math.round((slAtrMult * atrVal) / syminfo.mintick)
    float tpTicks = math.round((tpAtrMult * atrVal) / syminfo.mintick)
    strategy.entry("Short", strategy.short, qty=calculatedUnits, comment="SELL [Strat 1]")
    strategy.exit("Exit Short", "Short", loss=slTicks, profit=tpTicks, comment_loss="SL Hit", comment_profit="TP Hit")
    tradeSL := close + (slAtrMult * atrVal)
    tradeTP := close - (tpAtrMult * atrVal)
    alert("Strategy 1 SELL Signal triggered on " + syminfo.ticker, alert.freq_once_per_bar_close)

// Sync active SL/TP lines to exact fill price when open
if ta.change(strategy.position_size) != 0 and strategy.position_size != 0
    if strategy.position_size > 0
        tradeSL := strategy.position_avg_price - (slAtrMult * atrVal[1])
        tradeTP := strategy.position_avg_price + (tpAtrMult * atrVal[1])
    else if strategy.position_size < 0
        tradeSL := strategy.position_avg_price + (slAtrMult * atrVal[1])
        tradeTP := strategy.position_avg_price - (tpAtrMult * atrVal[1])

if strategy.position_size == 0
    tradeSL := na
    tradeTP := na

// =============================================================================
// 7. CHART PLOTS & VISUALS
// =============================================================================

// EMAs
plot(showEmas ? ema20 : na,  "20 EMA (Pullback Line)", color=color.new(#2962FF, 0), linewidth=2)
plot(showEmas ? ema50 : na,  "50 EMA (Trend Direction)", color=color.new(#FF6D00, 0), linewidth=2)
plot(showEmas ? ema100 : na, "100 EMA (Macro Baseline)", color=color.new(#9C27B0, 0), linewidth=3)

// Active SL & TP Lines
plot(showSLTP ? tradeSL : na, "Active Stop Loss", color=color.new(color.red, 0), style=plot.style_linebr, linewidth=2)
plot(showSLTP ? tradeTP : na, "Active Take Profit", color=color.new(color.green, 0), style=plot.style_linebr, linewidth=2)

// Session Background
bgcolor(showSessionBg and inSession ? color.new(#2196F3, 93) : na, title="London/NY Active Session Shading")

// Entry Shapes
plotshape(buySignal and canTradeToday and strategy.position_size == 0, "BUY Signal Marker", shape.triangleup, location.belowbar, color.new(color.lime, 0), size=size.small, text="BUY", textcolor=color.white)
plotshape(sellSignal and canTradeToday and strategy.position_size == 0, "SELL Signal Marker", shape.triangledown, location.abovebar, color.new(color.red, 0), size=size.small, text="SELL", textcolor=color.white)

// =============================================================================
// 8. ON-CHART REAL-TIME PROP METRICS DASHBOARD
// =============================================================================

var table dash = table.new(dashPosition, 2, 8, 
                           bgcolor=color.new(#131722, 10), 
                           border_color=color.new(color.gray, 60), 
                           border_width=1)

if showDashboard and barstate.islast
    // Determine status colors and texts cleanly without line continuation issues
    string statusText = "SESSION CLOSED"
    color statusCol = color.orange

    if targetPassed
        statusText := "TARGET PASSED (+8%)"
        statusCol := color.lime
    else if maxDrawdownHit
        statusText := "MAX DRAWDOWN HIT (-8%)"
        statusCol := color.red
    else if dailyLossBreaker
        statusText := "DAILY STOPPED (2 Losses/-$70)"
        statusCol := color.red
    else if dailyMaxTradesHit
        statusText := "DAILY TRADES LIMIT (3/3)"
        statusCol := color.orange
    else if inSession
        statusText := "ACTIVE (Scanning)"
        statusCol := color.green

    // Row 0: Header
    table.cell(dash, 0, 0, "Prop Bot [Strategy 1]", bgcolor=color.new(#1E222D, 0), text_color=color.white, text_size=size.small)
    table.cell(dash, 1, 0, "M15 Momentum Pullback", bgcolor=color.new(#1E222D, 0), text_color=color.aqua, text_size=size.small)

    // Row 1: Bot & Session Status
    table.cell(dash, 0, 1, "Status", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 1, statusText, bgcolor=color.new(#131722, 0), text_color=statusCol, text_size=size.small)

    // Row 2: Account Equity & Net PnL
    float currentEquity = propCapital + strategy.netprofit
    color pnlCol = strategy.netprofit >= 0 ? color.lime : color.red
    table.cell(dash, 0, 2, "Net Equity", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 2, "$" + str.tostring(currentEquity, "#.##") + " (" + (strategy.netprofit >= 0 ? "+" : "") + str.tostring(strategy.netprofit, "#.##") + ")", bgcolor=color.new(#131722, 0), text_color=pnlCol, text_size=size.small)

    // Row 3: Target Progress (+8% / $160)
    float targetPct = math.min(100.0, math.max(0.0, (strategy.netprofit / targetProfitUSD) * 100.0))
    table.cell(dash, 0, 3, "Target ($160)", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 3, str.tostring(targetPct, "#.#") + "%" + (targetPassed ? " [PASS]" : ""), bgcolor=color.new(#131722, 0), text_color=targetPassed ? color.lime : color.white, text_size=size.small)

    // Row 4: Daily PnL & Circuit Breaker
    table.cell(dash, 0, 4, "Daily PnL (Max -$70)", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 4, (dailyPnL >= 0 ? "+" : "") + "$" + str.tostring(dailyPnL, "#.##"), bgcolor=color.new(#131722, 0), text_color=dailyPnL >= 0 ? color.lime : color.red, text_size=size.small)

    // Row 5: Daily Losses & Trades Counter
    table.cell(dash, 0, 5, "Daily Losses / Trades", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 5, str.tostring(dailyLossesCount) + "/" + str.tostring(maxDailyLosses) + " L | " + str.tostring(dailyTradesCount) + "/" + str.tostring(maxDailyTrades) + " T", bgcolor=color.new(#131722, 0), text_color=dailyLossBreaker ? color.red : color.white, text_size=size.small)

    // Row 6: Indicators (Trend & RSI)
    string trendText = "NEUTRAL"
    color trendCol = color.orange
    if ema20 > ema50 and close > ema100
        trendText := "BULLISH"
        trendCol := color.lime
    else if ema20 < ema50 and close < ema100
        trendText := "BEARISH"
        trendCol := color.red
    table.cell(dash, 0, 6, "Trend & RSI(14)", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 6, trendText + " | RSI: " + str.tostring(rsiVal, "#.#"), bgcolor=color.new(#131722, 0), text_color=trendCol, text_size=size.small)

    // Row 7: Current ATR
    table.cell(dash, 0, 7, "ATR(14) Volatility", bgcolor=color.new(#131722, 0), text_color=color.gray, text_size=size.small)
    table.cell(dash, 1, 7, str.tostring(atrVal, "#.#####"), bgcolor=color.new(#131722, 0), text_color=color.white, text_size=size.small)
